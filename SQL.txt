-- 1. cria o banco
CREATE DATABASE IF NOT EXISTS aquario CHARACTER SET utf8mb4;

-- 2. cria o usuário que o programa Java vai usar
CREATE USER IF NOT EXISTS 'funcionario_cd'@'localhost' IDENTIFIED BY 'funcionario_pw';
GRANT ALL PRIVILEGES ON aquario.* TO 'funcionario_cd'@'localhost';
FLUSH PRIVILEGES;

-- 3. cria a tabela e alguns dados de exemplo
USE aquario;
CREATE TABLE IF NOT EXISTS animal (
    id    INT PRIMARY KEY AUTO_INCREMENT,
    nome  VARCHAR(80),
    especie VARCHAR(60) NOT NULL,
    tamanho DECIMAL(6,2),
    peso  DECIMAL(6,2),
    nascimento VARCHAR(10),
    dieta VARCHAR(30)
);
